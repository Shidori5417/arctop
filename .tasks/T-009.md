// app.rs
pub enum View { Overview, Charts, Procs, Help }          // 1/2/3, Tab döngüsü (Help hariç), F1 Help aç/kapa (önceki görünüme döner)
pub enum SortKey { Busy, Vram, Pid, Name }               // `s` döngüsü; Busy/Vram azalan, Pid/Name artan
pub enum ChartWindow { Last60s, Last5m }                 // `g`
pub enum Action { SetInterval(Duration) }                // main bunu sampler.set_interval'a iletir
pub struct App {
    pub view: View, pub sort: SortKey, pub window: ChartWindow,
    pub filter: String, pub filter_editing: bool,        // `/` düzenlemeyi açar; Enter/Esc kapatır (Esc filtreyi temizler)
    pub device: Device, pub capabilities: Vec<Capabilities>,
    pub last: Option<Snapshot>, pub history: History,
    pub interval: Duration,                              // `+`/`-`: 250 ms .. 5 s, ikiye katla/yarıya indir
    pub should_quit: bool,                               // `q` (filtre düzenlenirken q filtreye yazılır), Ctrl-C her zaman
}
impl App {
    pub fn new(device: Device, capabilities: Vec<Capabilities>, interval: Duration) -> Self;
    pub fn on_snapshot(&mut self, s: Snapshot);          // history.push + last
    pub fn on_key(&mut self, key: crossterm::event::KeyEvent) -> Option<Action>;
    /// Filtre + sıralama uygulanmış süreç listesi (procs görünümü kullanır).
    pub fn visible_procs(&self) -> Vec<&ProcUsage>;
    /// Testler için deterministik, dolu bir App (A750, tüm metrikler Value, bir-iki Unavailable, 5 süreç, 300 noktalı geçmiş).
    #[cfg(test)] pub fn sample() -> Self;
}

// ui/mod.rs
pub fn draw(f: &mut Frame, app: &App);   // üst: sekmeler (Özet/Grafikler/Süreçler) + kart adı; gövde: aktif görünüm; alt: kısayol satırı
// her görünüm dosyası: pub fn render(f: &mut Frame, area: Rect, app: &App)
// Bu kartta overview/charts/procs/help yalnızca başlıklı bir Block + "yakında" yazısı çizer (yer tutucu).

// ui/theme.rs — tüm renk/stiller burada: ACCENT (Arc mavisi, ör. Rgb(0x00,0x71,0xc5)), border, title, dim (Unavailable "—"), gauge renkleri (low/mid/high eşikleri), tabs stil fonksiyonları.

// fmt.rs — `fn metric<T: Display>(m: &Metric<T>, unit: &str) -> String` ("—" for Unavailable), `fn bytes(b: u64) -> String` (MiB/GiB), `fn pct(f32) -> String`.

## Davranış
- `main`: `SysFs::real()`, `discover`, ilk cihaz (yoksa anlaşılır hata, `anyhow`), kaynaklar Freq+Hwmon+Fdinfo, `Sampler::spawn(.., 1000 ms)`. CLI bu kartta yok (T-014 ekler); main'i T-014'ün bir `Config`/argüman yapısı ekleyebileceği şekilde küçük fonksiyonlara böl (`run_tui(dev, interval) -> anyhow::Result<()>`).
- Olay döngüsü: crossterm olayları ayrı bir thread'de okunup bir `crossbeam_channel`'a gönderilir; ana döngü `select!` ile olay ve snapshot kanallarını dinler (ARCHITECTURE §6). Çizim yalnızca değişiklikte (olay, snapshot, resize).
- Terminal: raw mode + alternate screen; çıkışta, hata dönüşünde **ve panikte** restore (panic hook önceki hook'u zincirler).
- Yeni crate yok (ratatui 0.29, crossterm 0.28 mevcut).
