```rust
// theme.rs
pub enum ColorMode { TrueColor, Ansi256, NoColor }
pub fn init(mode: ColorMode);                 // OnceLock; çağrılmazsa TrueColor (testler deterministik olsun)
pub fn detect_mode() -> ColorMode;           // NO_COLOR / COLORTERM kuralı (§2)
pub enum Tok { Accent, AccentBg, Text, Muted, Surface, RowAlt, Ok, Warn, Crit, Freq, Power, Temp, Busy, Vram }
pub fn color(t: Tok) -> Color;                // NoColor modunda Color::Reset
pub fn fg(t: Tok) -> Style; pub fn bold(t: Tok) -> Style;
pub fn load_color(pct: f32) -> Color;         // <50 Ok, <80 Warn, else Crit
pub fn temp_color(c: f32) -> Color;           // <60 Ok, <80 Warn, else Crit
pub fn selected() -> Style;                   // AccentBg zemin + bold (NoColor: REVERSED)
pub fn border() -> Style; pub fn border_focus() -> Style;

// widgets.rs  (hepsi saf; Frame almaz, widget/Line döndürür)
pub fn panel<'a>(title: &'a str) -> Block<'a>;                         // rounded, surface kenar, bold başlık, Padding::horizontal(1)
pub fn panel_with<'a>(title: &'a str, right: Line<'a>) -> Block<'a>;   // + sağ üst ikincil başlık
pub fn hbar(pct: f32, width: u16, color: Color) -> Line<'static>;     // ━ dolu (color) + ─ boş (Surface); pct 0..100 kırpılır
pub fn key_chips(keys: &[(&str, &str)]) -> Line<'static>;             // "tuş" Accent bold + " açıklama" Muted, aralarında 2 boşluk
pub fn big_value(value: &str, unit: &str, tok: Tok) -> Line<'static>; // bold metrik rengi + " birim" Muted
pub fn spark_data(ring: &Ring<Option<f64>>, width: usize) -> Vec<u64>;// son `width` nokta, None→0, ölçek: değerler ×10 yuvarlanır (Sparkline u64 ister)
pub fn centered(area: Rect, w: u16, h: u16) -> Rect;                  // modal için
pub fn empty_state<'a>(msg: &'a str) -> Paragraph<'a>;                // ortalanmış Muted metin

// fmt.rs ekleri
pub fn freq(mhz: u32) -> String;   // <1000 → "850 MHz", ≥1000 → "2.3 GHz"

// app.rs ekleri
pub fn help_return_view(&self) -> View;            // yardım açıkken arkada çizilecek görünüm
pub last_snapshot_at: Option<Instant>,             // on_snapshot'ta set edilir
pub fn is_fresh(&self, now: Instant) -> bool;      // last_snapshot_at + interval/2 > now
pub fn selected_pid(&self) -> Option<u32>;
pub fn selected_index(&self) -> Option<usize>;     // visible_procs() içindeki indeks
// on_key: Procs görünümünde ↑/↓, j/k, PgUp/PgDn (10), Home/End seçimi hareket ettirir (filtre düzenlenmiyorken).
//   Seçim PID'e bağlı; liste değişince aynı PID, yoksa en yakın indeks (on_snapshot sonrası da). Esc yardımı kapatır.
```

## ui/mod.rs (çerçeve, §3)
- Başlık satırı: ` ◆ arctop ` rozeti, `N Ad` sekme hapları (aktif `AccentBg`), sağda `cihaz · sürücü` (Muted) + canlılık noktası `●` (`is_fresh` → Ok, değilse Surface) + aralık (`1.0 s`). `draw` `Instant::now()` kullanır; testlerde deterministik olması için `draw_at(f, app, now)` iç fonksiyonu olsun.
- Altbilgi: `key_chips`, görünüme göre bağlamsal (§3). Filtre düzenleme satırı korunur.
- `View::Help`: önce `help_return_view()` görünümü gövdeye çizilir, sonra `help::render(f, body, app)` çağrılır — **help.rs modal'ı kendisi çizer** (bu kartta help.rs mevcut içeriğini `centered` + `Clear` + `panel` içine taşımak yeterli; tasarım T-029'da).
- Dar ekran kuralları (§3) ve 20x4'e kadar panik yok.
