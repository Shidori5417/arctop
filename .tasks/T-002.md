## Sabit arayüz (bu kart tanımlar; sonra değişmez)
```rust
// error.rs
#[derive(Debug, thiserror::Error)]
pub enum ReadError { NotFound(PathBuf), PermissionDenied(PathBuf), Parse { path: PathBuf, value: String }, Io { path: PathBuf, source: std::io::Error } }
impl ReadError { pub fn reason(&self) -> &'static str; } // Metric::Unavailable için kısa neden: "yok", "izin yok", "ayrıştırılamadı", "G/Ç hatası"

// sysfs.rs  — §1 aynen. Yollar mutlak verilir ("/sys/class/drm/card1/..."), root ile birleştirilir (baştaki "/" atılır).
pub struct SysFs { root: PathBuf }
impl SysFs {
    pub fn real() -> Self;                       // root = "/"
    pub fn with_root(root: impl Into<PathBuf>) -> Self;
    pub fn read_u64(&self, p: impl AsRef<Path>) -> Result<u64, ReadError>;      // trim; boşluktan sonraki birim yok sayılmaz → Parse hatası
    pub fn read_string(&self, p: impl AsRef<Path>) -> Result<String, ReadError>; // sondaki \n kırpılır
    pub fn read_dir(&self, p: impl AsRef<Path>) -> Result<Vec<PathBuf>, ReadError>; // dönen yollar root'suz mutlak yollar, sıralı
    pub fn read_link_name(&self, p: impl AsRef<Path>) -> Result<String, ReadError>; // symlink hedefinin son bileşeni; yoksa "<p>.name" dosyası
}

// model.rs
pub enum Metric<T> { Value(T), Unavailable(&'static str) }   // + Clone, Debug, PartialEq, serde::Serialize; yardımcılar: value(), is_available(), From<Result<T, ReadError>>
pub enum Driver { I915, Xe }                                   // + Display ("i915"/"xe"), FromStr
pub struct Device { pub card: u32, pub pci_slot: String, pub device_id: u16, pub name: String, pub driver: Driver }
pub enum EngineClass { Render, Copy, Video, VideoEnhance, Compute } // + fdinfo anahtar adı: as_fdinfo_key() -> "render" | "copy" | "video" | "video-enhance" | "compute"
pub struct EngineUsage { pub class: EngineClass, pub busy_pct: f32 }
pub struct FreqInfo { pub act: Metric<u32>, pub cur: Metric<u32>, pub min: Metric<u32>, pub max: Metric<u32>, pub boost: Metric<u32> }
pub struct ProcUsage { pub pid: u32, pub comm: String, pub engines: Vec<EngineUsage>, pub vram_bytes: Metric<u64> }
pub struct Snapshot { /* §4 alanları aynen */ }   // Snapshot::empty(at: Instant) tümünü Unavailable("örneklenmedi") ile kurar
pub struct Capabilities { pub fields: Vec<(&'static str, Metric<()>)> } // alan adı → okunabilir mi / neden değil
// Tümü Debug + Clone; Instant hariç Serialize (Snapshot.at için #[serde(skip)]).

// source/mod.rs — §3 trait'i aynen; alt modül bildirimleri (freq, hwmon, fdinfo) korunur.
pub trait MetricSource: Send { fn name(&self) -> &'static str; fn probe(&mut self, fs: &SysFs, dev: &Device) -> Capabilities; fn sample(&mut self, fs: &SysFs, dev: &Device, now: Instant, out: &mut Snapshot); }
```
`lib.rs` bu tipleri kök seviyede yeniden dışa aktarır (`pub use`).
